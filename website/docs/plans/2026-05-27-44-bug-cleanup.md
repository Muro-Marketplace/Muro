# 44-Bug Cleanup Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close all 44 bugs from the 2026-05-27 QA pass and Supabase audit, with validation harnesses that re-run after every phase so regressions get caught before merge.

**Architecture:** Fixes are grouped by root cause so each phase resolves a cluster of related bugs in one coherent migration / PR. RLS hardening lands first because it removes the underlying cause of 7 of the 11 critical/high paywall and impersonation findings. The audit harness (Phase 0) wraps the existing Supabase advisor MCP tool and Playwright `tests/e2e/smoke.spec.ts` into one `npm run audit:full` script that every later phase runs to verify the fix and prove nothing else regressed.

**Tech Stack:** Next.js 16.2, Supabase JS 2.103 (project `uwkuhygwvasdzwsusiym`, Postgres 17), Vitest 2.1 colocated `*.test.ts`, Playwright 1.49 in `tests/e2e/`, Stripe 22 (test mode), Tailwind 4. Path alias `@/` to `src/` (`tsconfig.json`).

**Bug list source of truth:** `/tmp/wallplace-bug-hunt-2026-05-27T20-56-55Z.md` (44 bugs, 7 critical / 16 high / 15 medium / 6 low).

**Branch + worktree strategy:** one worktree per phase under `.claude/worktrees/qa-44-<phase-letter>-<short-name>`. Each phase opens a draft PR after its first commit so reviewers can follow the changes incrementally. `npm run check` (lint + typecheck + vitest) must pass on every commit. Phase 0 must merge before any other phase begins; later phases can land in parallel.

```typescript
#!/usr/bin/env tsx
// Fetches the latest security + performance advisor lints for the
// Supabase project and writes them to scripts/audit/<security|performance>-current.json.
// Compare against the baseline to detect regressions.

import fs from "node:fs/promises";
import path from "node:path";

const PROJECT_REF = "uwkuhygwvasdzwsusiym";
const TOKEN = process.env.SUPABASE_ACCESS_TOKEN;
if (!TOKEN) {
  console.error("SUPABASE_ACCESS_TOKEN not set");
  process.exit(2);
}

async function fetchLints(type: "security" | "performance") {
  const url = `https://api.supabase.com/v1/projects/${PROJECT_REF}/database/lints?type=${type}`;
  const res = await fetch(url, {
    headers: { Authorization: `Bearer ${TOKEN}` },
  });
  if (!res.ok) {
    console.error(`${type} fetch failed: ${res.status} ${await res.text()}`);
    process.exit(2);
  }
  return (await res.json()) as Array<{ name: string; metadata?: { name?: string }; detail: string }>;
}

async function main() {
  const outDir = path.join(process.cwd(), "scripts", "audit");
  for (const type of ["security", "performance"] as const) {
    const lints = await fetchLints(type);
    const file = path.join(outDir, `${type}-current.json`);
    await fs.writeFile(file, JSON.stringify(lints, null, 2));
    console.log(`Wrote ${lints.length} ${type} lints to ${file}`);
  }
}

main().catch((e) => {
  console.error(e);
  process.exit(2);
});
```

```typescript
#!/usr/bin/env tsx
import fs from "node:fs/promises";
import path from "node:path";

type Lint = { cache_key: string; [k: string]: unknown };

export function findNewLints(baseline: Lint[], current: Lint[]): Lint[] {
  const seen = new Set(baseline.map((l) => l.cache_key));
  return current.filter((l) => !seen.has(l.cache_key));
}

async function main() {
  const dir = path.join(process.cwd(), "scripts", "audit");
  const baseline: Lint[] = JSON.parse(await fs.readFile(path.join(dir, "baseline-advisors.json"), "utf8"));
  const current: Lint[] = JSON.parse(await fs.readFile(path.join(dir, "security-current.json"), "utf8"));
  const added = findNewLints(baseline, current);
  if (added.length) {
    console.error(`\nFAIL: ${added.length} new advisor lint(s) introduced since baseline:\n`);
    for (const l of added) {
      console.error(`  - ${l.name ?? "?"}: ${l.detail ?? l.cache_key}`);
    }
    console.error("\nIf this lint is intentional, add it to baseline-advisors.json.\n");
    process.exit(1);
  }
  console.log(`PASS: no new advisor lints (${current.length} total, ${baseline.length} in baseline)`);
}

if (import.meta.url === `file://${process.argv[1]}`) {
  main().catch((e) => {
    console.error(e);
    process.exit(2);
  });
}
```

```sql
-- 070_orders_tighten_rls.sql
--
-- Bug 28: drop the `Users can read their orders` policy (qual: true)
--   which let every authenticated user read every order.
-- Bug 29: drop the three INSERT (WITH CHECK true) policies on orders.
--   Orders are created server-side after the Stripe webhook fires; the
--   service-role client bypasses RLS, so no anon/authenticated INSERT
--   policy is needed.

DROP POLICY IF EXISTS "Users can read their orders" ON public.orders;
DROP POLICY IF EXISTS "Users can read own orders"   ON public.orders;
DROP POLICY IF EXISTS "Allow anon inserts"          ON public.orders;
DROP POLICY IF EXISTS "Allow authenticated inserts" ON public.orders;
DROP POLICY IF EXISTS "Service role can insert orders" ON public.orders;

-- The narrow SELECT policies orders_select_buyer, orders_select_artist,
-- orders_select_venue are already correct and stay in place.

NOTIFY pgrst, 'reload schema';
```

```typescript
// src/lib/checkout/check-seller-ready.ts
import { stripe } from "@/lib/stripe";
import { supabaseAdmin } from "@/lib/db/supabase-admin";

export type SellerReadiness =
  | { ready: true }
  | { ready: false; reason: "no_stripe_account" | "payouts_not_enabled"; sellerName: string };

export async function checkSellerReady(artistUserId: string): Promise<SellerReadiness> {
  const { data: profile } = await supabaseAdmin
    .from("artist_profiles")
    .select("display_name, stripe_account_id")
    .eq("user_id", artistUserId)
    .single();
  if (!profile?.stripe_account_id) {
    return { ready: false, reason: "no_stripe_account", sellerName: profile?.display_name ?? "the artist" };
  }
  const acct = await stripe.accounts.retrieve(profile.stripe_account_id);
  if (!acct.charges_enabled) {
    return { ready: false, reason: "payouts_not_enabled", sellerName: profile.display_name };
  }
  return { ready: true };
}
```
